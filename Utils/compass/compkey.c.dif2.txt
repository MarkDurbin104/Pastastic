 204: SORTRPLU *SortList(REMACSPLU *plulist,BOOL by_code)
 205: {
 206:     SORTRPLU *firstelem=NULL;
 207:     SORTRPLU *traverse;
 208:     SORTRPLU *lastelem;
 209:     SORTRPLU *curelem;
 210:     BOOL additem;
 211: 
 212:     while (plulist!=NULL) {
 213:         if ((curelem=malloc(sizeof(SORTRPLU)))==NULL) return NULL;
 214:         curelem->elem=plulist;
 215:         if (firstelem==NULL) {
 216:             firstelem=curelem;
 217:             firstelem->next=NULL;
 218:         } else {
 219:             traverse=firstelem;
 220:             lastelem=NULL;
 221:             while (traverse!=NULL) {
 222:                 if (by_code) {
 223:                     additem=(curelem->elem->plu_code<traverse->elem->plu_code);
 224:                 } else {
 225:                     additem=(strcmp(curelem->elem->desc,traverse->elem->desc)<=0);
 226:                 }
 227:                 if (additem) {
 228:                     if (lastelem==NULL) {
 229:                         curelem->next=traverse;
 230:                         firstelem=curelem;
 231:                     } else {
 232:                         curelem->next=traverse;
 233:                         lastelem->next=curelem;
 234:                     }
 235:                     break;
 236:                 } else {
 237:                     if (traverse->next==NULL) {
 238:                         curelem->next=NULL;
 239:                         traverse->next=curelem;
 240:                         break;
 241:                     }
 242:                 }
 243:                 lastelem=traverse;
 244:                 traverse=traverse->next;
 245:             }
 246:         }
 247:         plulist=plulist->next;
 248:     }
 249:     return firstelem;
 250: }

 360: BOOL isvalid(INT *key,CHAR type)
 361: {
 362:     if ((*key<32)||(*key>126)) return FALSE;
 363:     switch (type) {
<INSERT2>
 364:         case 'D' :
 365:             if (isdigit(*key)||(*key=='/')) return TRUE;
 366:             break;
 367:         case '.' :
 368:             if (isalnum(*key)||(*key=='\\')||(*key=='.')) return TRUE;
 369:             break;
<END>
 370:         case 'X' :
 371:             if (isalnum(*key)||(*key==' ')) return TRUE;
 372:             break;
 373:         case '9' :
 374:             if (isdigit(*key)) return TRUE;
 375:             break;
 376:         case 'Y' :
 377:             if ((toupper(*key)=='Y')||(toupper(*key)=='N')) {
 378:                 *key=toupper(*key);
 379:                 return TRUE;
 380:             }
 381:             break;
 382:     }
 383:     return FALSE;
 384: }

 391: //P The attribute

 401: CHAR *InputLine(MAINDS *ds,INT x,INT y,INT attr,CHAR *str,INT maxwid,INT cx,INT cy,INT *lastkey,CHAR type)
 402: {
 403:     static CHAR buffer[MAXSTRLEN];
 404:     INT cursor;
 405:     INT key;
 406:     BOOL xit=FALSE;
 407: 
 408:     if (str==NULL)
 409:         buffer[0]='\0';
 410:     else
 411:         strcpy(buffer,str);
 412: 
 413:     cursor=strlen(buffer);
 414:     while (!xit) {
 415:         OutFormat(x,y,attr,"%-*.*s",maxwid,maxwid,buffer);
 416:         LargeCursor();
<INSERT2>
 417:         PlaceCursor(x+cursor+1,y+1);
 418: //        gotoxy(x+cursor+1,y+1);
 419:         if (cx>=0) {
 420:             KeyCursor(GRIDLEFT+1+cx*ds->keys.width,GRIDTOP+1+cy*ds->keys.height,
<END>
 421:                   ds->keys.width-1,ds->keys.height-1,G_colour[0]);
<INSERT2>
 422:         }
<END>
 423:         key=GetKey();
 424:         HideCursor();
 425:         switch (key) {
 426:             case BKSP :
 427:                 if (cursor>0) cursor--;
 428:                 buffer[cursor]='\0';
 429:                 break;
 430:             case ESC :
 431:                 if (str==NULL)
 432:                     buffer[0]='\0';
 433:                 else
 434:                     strcpy(buffer,str);
 435:                 cursor=strlen(buffer);
 436:                 xit=1;
 437:                 break;
 438:             case STAB :
 439:             case TAB :
 440:             case ENTER :
 441:             case F01 :
 442:             case F02 :
 443:             case PGDN:
 444:             case PGUP:
 445:             case INS :
 446:             case DEL :
 447:                 xit=TRUE;
 448:                 break;
 449:             case F09 :
 450:                 RunHelp(EDITHELP);
 451:                 break;
 452:             default :
 453:                 if ((cursor<maxwid)&&(isvalid(&key,type))) {
 454:                     buffer[cursor++]=(CHAR)key;
 455:                     buffer[cursor]='\0';
 456:                 } else {
<INSERT2>
 457:                     Beep();
<END>
<SPACE>


<END>
 458:                 }
 459:         }
 460:     }
 461:     *lastkey=key;
 462:     return buffer;

 538:         }
 539:         oddeve^=1;
 540:         curkey++;
 541:     }
