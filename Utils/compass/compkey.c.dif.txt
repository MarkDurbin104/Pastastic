 173: SORTRPLU *SortList(REMACSPLU *plulist,BOOL by_code)
 174: {
 175:     SORTRPLU *firstelem=NULL;
 176:     SORTRPLU *traverse;
 177:     SORTRPLU *lastelem;
 178:     SORTRPLU *curelem;
 179:     BOOL additem;
 180: 
 181:     while (plulist!=NULL) {
 182:         if ((curelem=malloc(sizeof(SORTRPLU)))==NULL) return NULL;
 183:         curelem->elem=plulist;
 184:         if (firstelem==NULL) {
 185:             firstelem=curelem;
 186:             firstelem->next=NULL;
 187:         } else {
 188:             traverse=firstelem;
 189:             lastelem=NULL;
 190:             while (traverse!=NULL) {
 191:                 if (by_code) {
 192:                     additem=(curelem->elem->plu_code<traverse->elem->plu_code);
 193:                 } else {
 194:                     additem=(strcmp(curelem->elem->desc,traverse->elem->desc)<=0);
 195:                 }
 196:                 if (additem) {
 197:                     if (lastelem==NULL) {
 198:                         curelem->next=traverse;
 199:                         firstelem=curelem;
 200:                     } else {
 201:                         curelem->next=traverse;
 202:                         lastelem->next=curelem;
 203:                     }
 204:                     break;
 205:                 } else {
 206:                     if (traverse->next==NULL) {
 207:                         curelem->next=NULL;
 208:                         traverse->next=curelem;
 209:                         break;
 210:                     }
 211:                 }
 212:                 lastelem=traverse;
 213:                 traverse=traverse->next;
 214:             }
 215:         }
 216:         plulist=plulist->next;
 217:     }
 218:     return firstelem;
 219: }

 225: void RunSearch(MAINDS *ds,LONG *selected_plu)
 226: {
 227:     static UINT scrbuffer[25*80];
 228:     static SORTRPLU *code_list=NULL; 
 229:     static SORTRPLU *desc_list=NULL;
 230:     SORTRPLU *curlist;
 231:     BOOL xit=FALSE;
 232:     INT key;
 233:     INT topline=0;
 234:     INT curline=0;
 235:     INT numlines;
 236:     LONG curplu;
 237: 
 238:     memcpy(scrbuffer,((UINT far *)SCREENBASE),25*80*2);
 239: 
 240:     StatusLine(G_tsearch,0);
 241: 
 242:     SetViewPort(0,0,79,21,G_colour[18]);
 243:     Rectangle(0,0,79,21,G_colour[17],0);
 244: 
 245:     if (code_list==NULL) code_list=SortList(ds->rplulist,TRUE);
 246:     if (desc_list==NULL) desc_list=SortList(ds->rplulist,FALSE);
 247: 
 248:     curlist=code_list;
 249:     if (curlist==NULL) return;
 250: 
 251:     while (!xit) {
 252:         numlines=ListPLULines(1,1,19,topline,curline,curlist,&curplu);
 253:         key=GetKey();
 254:         switch (key) {
 255:             case F09 :
 256:                 RunHelp(SEARCHHELP1);
 257:                 break;
 258:             case STAB :
 259:             case UP :
 260:                 if (curline>topline) {
 261:                     curline--;
 262:                 } else {
 263:                     if (topline>0) {
 264:                         topline--;
 265:                         curline--;
 266:                     }
 267:                 }
 268:                 break;
 269:             case PGUP :
 270:                 if (topline>19) {
 271:                     topline-=19;
 272:                     curline-=19;
 273:                 } else {
 274:                     topline=0;
 275:                     curline=0;
 276:                 }
 277:                 break;
 278:             case TAB :
 279:             case DOWN :
 280:                 if (curline<numlines-1) {
 281:                     if (curline>=topline+19) {
 282:                         topline++;
 283:                     }
 284:                     curline++;
 285:                 }
 286:                 break;
 287:             case PGDN :
 288:                 if (topline<numlines-20) {
 289:                     topline+=19;
 290:                     curline+=19;
 291:                 } else {
 292:                     topline=numlines-20;
 293:                     curline=numlines-1;
 294:                 }
 295:                 break;
 296:             case ENTER :
 297:                 *selected_plu=curplu;
 298:                 xit=TRUE;
 299:                 break;
 300:             case ESC :
 301:                 xit=TRUE;
 302:                 break;
 303:             default :
 304:                 if ((key>31)&&(key<126)) {
 305:                     curlist=desc_list;
 306:                     topline=0;
 307:                     while (curlist!=NULL) {
 308:                         if (toupper(curlist->elem->desc[0])==toupper(key)) {
 309:                             break;
 310:                         }
 311:                         topline++;
 312:                         curlist=curlist->next;
 313:                     }
 314:                     if (curlist==NULL) {
 315:                         topline=0;
<INSERT1>
 316:                         sound(1000);
 317:                         delay(100);
 318:                         nosound();
<END>
 319:                     }
 320:                     curlist=desc_list;
 321:                     curline=topline;
 322:                 } else {
<INSERT1>
 323:                     sound(1000);
 324:                     delay(100);
 325:                     nosound();
<END>
 326:                 }
 327:         }
 328:     }
 329:     SetViewPort(0,0,80,25,0);
 330:     memcpy(((UINT far *)SCREENBASE),scrbuffer,25*80*2);
 331: }

 333: BOOL isvalid(INT *key,CHAR type)
 334: {
 335:     if ((*key<32)||(*key>126)) return FALSE;
 336:     switch (type) {
<SPACE>






<END>
 337:         case 'X' :
 338:             if (isalnum(*key)||(*key==' ')) return TRUE;
 339:             break;
 340:         case '9' :
 341:             if (isdigit(*key)) return TRUE;
 342:             break;
 343:         case 'Y' :
 344:             if ((toupper(*key)=='Y')||(toupper(*key)=='N')) {
 345:                 *key=toupper(*key);
 346:                 return TRUE;
 347:             }
 348:             break;
 349:     }
 350:     return FALSE;
 351: }

 353: //F InputLine
 354: //S Input a character string
 355: //D Input a character string
 356: //P Zero based X co-ordinate (origin Top left)
 357: //P Zero based Y co-ordinate (origin Top left)
 358: //P The attribute
 359: //P The string to start with
 360: //P The maximum width of the input string
 361: //P The current keyboard cursor x co-ordinate (the flashing cursor)
 362: //P The current keyboard cursor y co-ordinate 
 363: //P The current keyboard cursor width 
 364: //P The current keyboard cursor height 
 365: //P The last key pressed (the key that caused exit)
 366: //P input format type 'X' alphanum '9' numeric 'Y' yes/no
 367: //R The input string
 368: CHAR *InputLine(MAINDS *ds,INT x,INT y,INT attr,CHAR *str,INT maxwid,INT cx,INT cy,INT *lastkey,CHAR type)
 369: {
 370:     static CHAR buffer[MAXSTRLEN];
 371:     INT cursor;
 372:     INT key;
 373:     BOOL xit=FALSE;
 374: 
 375:     if (str==NULL)
 376:         buffer[0]='\0';
 377:     else
 378:         strcpy(buffer,str);
 379: 
 380:     cursor=strlen(buffer);
 381:     while (!xit) {
 382:         OutFormat(x,y,attr,"%-*.*s",maxwid,maxwid,buffer);
 383:         LargeCursor();
<SPACE>


<END>
<INSERT1>
 384:         gotoxy(x+cursor+1,y+1);
 385:         KeyCursor(GRIDLEFT+1+cx*ds->keys.width,GRIDTOP+1+cy*ds->keys.height,
<END>
 386:                   ds->keys.width-1,ds->keys.height-1,G_colour[0]);
<SPACE>

<END>
 387:         key=GetKey();
 388:         HideCursor();
 389:         switch (key) {
 390:             case BKSP :
 391:                 if (cursor>0) cursor--;
 392:                 buffer[cursor]='\0';
 393:                 break;
 394:             case ESC :
 395:                 if (str==NULL)
 396:                     buffer[0]='\0';
 397:                 else
 398:                     strcpy(buffer,str);
 399:                 cursor=strlen(buffer);
 400:                 xit=1;
 401:                 break;
 402:             case STAB :
 403:             case TAB :
 404:             case ENTER :
 405:             case F01 :
 406:             case F02 :
 407:             case PGDN:
 408:             case PGUP:
 409:             case INS :
 410:             case DEL :
 411:                 xit=TRUE;
 412:                 break;
 413:             case F09 :
 414:                 RunHelp(EDITHELP);
 415:                 break;
 416:             default :
 417:                 if ((cursor<maxwid)&&(isvalid(&key,type))) {
 418:                     buffer[cursor++]=(CHAR)key;
 419:                     buffer[cursor]='\0';
 420:                 } else {
<INSERT1>
 421:                     sound(1000);
 422:                     delay(100);
 423:                     nosound();
<END>
 424:                 }
 425:         }
 426:     }
 427:     *lastkey=key;
 428:     return buffer;
 429: }

 489: void StatusLine(CHAR **keys,BOOL flash)
 490: {
 491:     CHAR **curkey;
 492:     INT oddeve=0;
 493:     INT dwid=0;
 494:     INT fullwid=-1;
 495:     INT curofs;
 496: 
 497:     OutFormat(1,23,G_colour[13],"%*.*s",78,78,"");
 498:     curkey=keys;
 499:     while (*curkey!=NULL) {
 500:         if (oddeve) {
 501:             fullwid+=dwid+strlen(*curkey)+1;
 502:         } else {
 503:             dwid=strlen(*curkey)+1;
 504:         }
 505:         oddeve^=1;
 506:         curkey++;
 507:     }                           
 508:     curkey=keys;                
 509:     oddeve=0;                   
 510:     curofs=(78-fullwid)/2;      
 511:     while (*curkey!=NULL) {
 512:         if (oddeve) {
 513:             if (flash)
 514:                 OutFormat(curofs+dwid,23,G_colour[12]|0x80,*curkey);
 515:             else 
 516:                 OutFormat(curofs+dwid,23,G_colour[12],*curkey);

 518:         } else {
 519:             dwid=strlen(*curkey)+1;
 520:             if (flash)
 521:                 OutFormat(curofs,23,G_colour[11]|0x80,"%s ",*curkey);
 522:             else 
 523:                 OutFormat(curofs,23,G_colour[11],"%s ",*curkey);
 524:         }
 525:         oddeve^=1;
 526:         curkey++;
 527:     }
