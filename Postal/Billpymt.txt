Public Function GetStatusCode(status As Long) As String
On Error GoTo ErrHandler
   
   Dim msg As String
   Select Case status
      Case IP_SUCCESS:               msg = "success"
      Case INADDR_NONE:              msg = "inet_addr: bad IP format"
      Case IP_BUF_TOO_SMALL:         msg = "buf too_small"
      Case IP_DEST_NET_UNREACHABLE:  msg = "dest net unreachable"
      Case IP_DEST_HOST_UNREACHABLE: msg = "dest host unreachable"
      Case IP_DEST_PROT_UNREACHABLE: msg = "dest prot unreachable"
      Case IP_DEST_PORT_UNREACHABLE: msg = "dest port unreachable"
      Case IP_NO_RESOURCES:          msg = "no resources"
      Case IP_BAD_OPTION:            msg = "bad option"
      Case IP_HW_ERROR:              msg = "hw_error"
      Case IP_PACKET_TOO_BIG:        msg = "packet too_big"
      Case IP_REQ_TIMED_OUT:         msg = "req timed out"
      Case IP_BAD_REQ:               msg = "bad req"
      Case IP_BAD_ROUTE:             msg = "bad route"
      Case IP_TTL_EXPIRED_TRANSIT:   msg = "ttl expired transit"
      Case IP_TTL_EXPIRED_REASSEM:   msg = "ttl expired reassem"
      Case IP_PARAM_PROBLEM:         msg = "param_problem"
      Case IP_SOURCE_QUENCH:         msg = "source quench"
      Case IP_OPTION_TOO_BIG:        msg = "option too_big"
      Case IP_BAD_DESTINATION:       msg = "bad destination"
      Case IP_ADDR_DELETED:          msg = "addr deleted"
      Case IP_SPEC_MTU_CHANGE:       msg = "spec mtu change"
      Case IP_MTU_CHANGE:            msg = "mtu_change"
      Case IP_UNLOAD:                msg = "unload"
      Case IP_ADDR_ADDED:            msg = "addr added"
      Case IP_GENERAL_FAILURE:       msg = "general failure"
      Case IP_PENDING:               msg = "pending"
      Case PING_TIMEOUT:             msg = "ping timeout"
      Case Else:                     msg = "unknown  msg returned"
   End Select
   GetStatusCode = msg
    
    Exit Function
ErrHandler:
    VError = "Error " & Err
    VErrorStr = Error$
    VErrorSub = "GetStatusCode"
    VerrorLocation = "ModuleBillServer"
    ErrorLog
    Resume Next
   
End Function

Public Function CanPing(ByVal s_serverIP As String) As Boolean
On Error GoTo ErrHandler
    
    Dim ECHO As ICMP_ECHO_REPLY
    Dim pos As Long
    Dim success As Long
    If SocketsInitialize() Then
        success = Ping((s_serverIP), ("PAPing"), ECHO)
        If GetStatusCode(success) = "success" Then
            CanPing = True
        Else
            CanPing = False
        End If
        SocketsCleanup
    Else

    End If
    
    Exit Function
ErrHandler:
    VError = "Error " & Err
    VErrorStr = Error$
    VErrorSub = "CanPing"
    VerrorLocation = "ModuleBillServer"
    ErrorLog
    Resume Next
    
End Function
Public Function Ping(sAddress As String, sDataToSend As String, ECHO As ICMP_ECHO_REPLY) As Long
On Error GoTo ErrHandler
  
  'If Ping succeeds :
  '.RoundTripTime = time in ms for the ping to complete,
  '.Data is the data returned (NULL terminated)
  '.Address is the Ip address that actually replied
  '.DataSize is the size of the string in .Data
  '.Status will be 0
  '
  'If Ping fails .Status will be the error code
   Dim hPort As Long
   Dim dwAddress As Long
  'convert the address into a long representation
   dwAddress = inet_addr(sAddress)
  'if a valid address..
   If dwAddress <> INADDR_NONE Then
     'open a port
      hPort = IcmpCreateFile()
     'and if successful,
      If hPort Then
        'ping it.
         Call IcmpSendEcho(hPort, dwAddress, sDataToSend, Len(sDataToSend), 0, ECHO, Len(ECHO), PING_TIMEOUT)
        'return the status as ping succes and close
         Ping = ECHO.status
         Call IcmpCloseHandle(hPort)
      End If
   Else:
        'the address format was probably invalid
         Ping = INADDR_NONE
   End If
    
    Exit Function
ErrHandler:
    VError = "Error " & Err
    VErrorStr = Error$
    VErrorSub = "Ping"
    VerrorLocation = "ModuleBillServer"
    ErrorLog
    Resume Next
   
End Function

Public Sub OperatorBalance()
On Error GoTo ErrHandler
    Dim FileNum1 As Single
    Dim TotalCash As Double
    Dim ErrAmount As Double
    Dim AppCash As String
    Dim cn As New rdoConnection
    Dim qd As New rdoQuery
    Dim rs As rdoResultset
    Dim PayDate As String
    Dim HowLong As Integer
    Dim CashAmount As String
    
    If CanPing(s_serverIP) = True Then
        Screen.MousePointer = vbHourglass
        cn.CursorDriver = rdUseOdbc
        cn.Connect = "Driver={" & s_driver & "};Server=" & s_serverName & ";UID=" & s_userID & ";PWD=" & s_password & ";Database=" & s_databaseName & ";"
        cn.EstablishConnection
        With qd
            Set .ActiveConnection = cn
            .SQL = "Select * From bill_data_invoice where op_id='" & UCase(VOp_ID) & "'"
            .RowsetSize = 1
            Set rs = .OpenResultset(rdOpenKeyset, rdConcurRowVer)
        End With
        If rs.RowCount = 0 Then
            cn.Close
            MsgBox "No Transaction Data Found    ", vbCritical
            Exit Sub
        End If
        GotOne = False
        FileNum1 = FreeFile
        Open s_home_dir & "\work\" & VOp_ID & "Balance" & Format$(Date, "ddmmmyyyy") & ".txt" For Append As #FileNum1
        Print #FileNum1, "Q-Post Operator Balance " & VOffice & Space(1) & Format$(Date, "dd mmmm yyyy") & Space(1) & Format$(Time, "hh:mm:ss")
        Print #FileNum1, Space(1)
        Print #FileNum1, " Operator: " & UCase(VOp_ID)
        Print #FileNum1, Space(1)
        Do While Not rs.EOF
            With rs
                PayDate = Mid$(!payment_date, 1, 2) & "/" & Mid$(!payment_date, 3, 2) & "/" & Mid$(!payment_date, 5, 4)
                If !update_sent = False And !cancelled = False And PayDate = Format(Date, "dd/mm/yyyy") Then
                    CashAmount = Format$(!total_amount, "#########.00")
                    PayDate = Mid$(!payment_date, 1, 2) & "/" & Mid$(!payment_date, 3, 2) & "/" & Mid$(!payment_date, 5, 4)
                    HowLong = 10 - Len(CashAmount)
                    Print #FileNum1, CashAmount & Space(HowLong + 1) & "Bill Ref: " & !customer_number & Space(1) & "Receipt No: " & !qpost_reference & Space(2) & PayDate & Space(2) & !op_id
                    TotalCash = TotalCash + Val(CashAmount)
                    GotOne = True
                    .Edit
                        !op_balance = True
                    .Update
                End If
                .MoveNext
            End With
        Loop
        If GotOne = False Then
            cn.Close
            Close #FileNum1
            MsgBox "No Transaction Data Available    ", vbCritical
            Screen.MousePointer = vbDefault
            Exit Sub
        End If
        rs.MoveFirst
        Print #FileNum1, Space(1)
        Print #FileNum1, "Cancelled Receipts " & VOffice
        Print #FileNum1, Space(1)
        Do While Not rs.EOF
                With rs
                PayDate = Mid$(!payment_date, 1, 2) & "/" & Mid$(!payment_date, 3, 2) & "/" & Mid$(!payment_date, 5, 4)
                If !update_sent = False And !cancelled = True And GotOne = True And PayDate = Format(Date, "dd/mm/yyyy") Then
                    CashAmount = Format$(!total_amount, "#########.00")
                    'PayDate = Mid$(!payment_date, 1, 2) & "/" & Mid$(!payment_date, 3, 2) & "/" & Mid$(!payment_date, 5, 4)
                    HowLong = 10 - Len(CashAmount)
                    Print #FileNum1, "Bill Ref: " & !customer_number & Space(1) & "Receipt No: " & !qpost_reference & Space(2) & PayDate & Space(2) & !op_id
                    ErrAmount = ErrAmount + Val(!total_amount)
                End If
                'CashAmount & Space(HowLong + 1) &
                .MoveNext
            End With
        Loop
        TotalCash = TotalCash ' - ErrAmount
        Print #FileNum1, Space(1)
        Print #FileNum1, "Grand Total: QR "; Format$(TotalCash, "#########.00")
        Close #FileNum1
        cn.Close
        FileNum1 = FreeFile
        Open s_home_dir & "\work\" & VOp_ID & "Balance" & Format$(Date, "ddmmmyyyy") & ".txt" For Input As #FileNum1
        Printer.Font.Name = "Arial"
        Printer.Font.Size = 10
        Do While Not EOF(FileNum1)
            Input #FileNum1, AppCash
            Printer.Print Space(8) & AppCash
        Loop
        Printer.Print Space(1)
        Printer.EndDoc
        Close #FileNum1
        Kill s_home_dir & "\work\" & VOp_ID & "Balance" & Format$(Date, "ddmmmyyyy") & ".txt"
        Screen.MousePointer = vbDefault
    Else
        MsgBox "Network problem or the server is unavailable, please call IT Support   ", vbCritical
    End If
        
    Exit Sub
    
ErrHandler:
    MsgBox "Error " & Err & ": " & Error$, vbMsgBoxRtlReading, "ModuleBills - ProcessEmailAttachment"
    VError = "Error " & Err & ": " & Error$ & " in ModuleBills, OperatorBalance"
    Call Err_email
    Resume Next

End Sub

Public Sub ErrorLog()
On Error GoTo ErrHandler
    Dim cn As New rdoConnection
    Dim qd As New rdoQuery
    Dim rs As rdoResultset
    
    If CanPing(s_serverIP) = True Then
        Screen.MousePointer = vbHourglass
        cn.CursorDriver = rdUseOdbc
        cn.Connect = "Driver={" & s_driver & "};Server=" & s_serverName & ";UID=" & s_userID & ";PWD=" & s_password & ";Database=" & s_databaseName & ";"
        cn.EstablishConnection
        With qd
            Set .ActiveConnection = cn
            .SQL = "Select * From errorlog"
            .RowsetSize = 1
            Set rs = .OpenResultset(rdOpenKeyset, rdConcurRowVer)
        End With
        With rs
            .AddNew
            !terminalID = s_serverIP
            !errorDate = Date
            !errorTime = Time
            !errornumber = VError
            !errordescription = VErrorStr
            !errorlocation = VerrorLocation
            !errorprocedure = VErrorSub
            .Update
        End With
        cn.Close
    End If
    Screen.MousePointer = vbDefault

    Exit Sub
ErrHandler:
    VError = "Error " & Err
    VErrorStr = Error$
    VErrorSub = "ErrorLog"
    VerrorLocation = "ModuleBillServer"
    ErrorLog
    Resume Next

End Sub
